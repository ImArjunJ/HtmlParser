cmake_minimum_required(VERSION 3.16)
project(HtmlParserConsumer LANGUAGES CXX)
if(HTMLPARSER_SOURCE_DIR)
    add_subdirectory("${HTMLPARSER_SOURCE_DIR}" HtmlParser)
    if(HTMLPARSER_BUILD_TESTS OR HTMLPARSER_BUILD_EXAMPLES)
        message(FATAL_ERROR "Embedded builds must disable optional targets by default")
    endif()
else()
    find_package(HtmlParser 0.2 REQUIRED CONFIG)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE HtmlParser::HtmlParser)
enable_testing()
add_test(NAME consumer COMMAND consumer)
